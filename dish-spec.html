ESP32 Firmware — Core Logic
// River Pointe — Station Sensor Firmware // Flash to ESP32 via Arduino IDE or PlatformIO // Library: HX711 by bogde (Library Manager) #include "HX711.h" #include "WiFi.h" #include "HTTPClient.h" // ── CONFIG ────────────────────────── const char* SSID = "RPI_KITCHEN"; const char* PASS = "your_wifi_password"; const char* WORKER_URL = "https://rpi-cutlery.workers.dev/event"; const char* API_KEY = "your_api_key"; const char* STATION_ID = "PASS"; // Calibration: grams per raw unit // Run calibration sketch first with // known weights to get these values float CALIBRATION[4] = {420.0, 415.0, 412.0, 418.0}; // Piece weight profiles (grams each) float PIECE_WEIGHT[4] = {85.0, 95.0, 110.0, 85.0}; // Bin codes match Airtable exactly const char* BIN_CODES[4] = { "PASS-DFORK-1", "PASS-DKNIFE-1", "PASS-SKNIFE-1", "PASS-SFORK-1" }; // GPIO pins (DT, SCK) per HX711 int DT_PINS[4] = {4, 18, 21, 23}; int SCK_PINS[4] = {5, 19, 22, 25}; HX711 scales[4]; int lastQty[4] = {-1,-1,-1,-1}; const long READ_INTERVAL = 30000; // 30s const int CHANGE_THRESH = 1; // min piece delta void setup() { Serial.begin(115200); for (int i=0; i<4; i++) { scales[i].begin(DT_PINS[i], SCK_PINS[i]); scales[i].set_scale(CALIBRATION[i]); scales[i].tare(); } WiFi.begin(SSID, PASS); while (WiFi.status() != WL_CONNECTED) delay(500); } void loop() { for (int i=0; i<4; i++) { float grams = scales[i].get_units(5); int qty = max(0, (int)(grams / PIECE_WEIGHT[i])); if (abs(qty - lastQty[i]) >= CHANGE_THRESH) { postReading(BIN_CODES[i], qty); lastQty[i] = qty; } } delay(READ_INTERVAL); } void postReading(const char* bin, int qty) { HTTPClient http; http.begin(WORKER_URL); http.addHeader("Content-Type", "application/json"); http.addHeader("X-API-Key", API_KEY); String body = "{" "\"sensor_type\":\"SENSOR_WEIGHT\"," "\"bin_code\":\"" + String(bin) + "\"," "\"quantity\":" + String(qty) + "}"; http.POST(body); http.end(); }
Calibration — One-Time Setup Per Bin
Step 1 — Tare
Place the empty bin on the platform. Power on the ESP32. Run the tare function — this zeros out the platform + bin weight so only the cutlery weight is measured.
Step 2 — Known Weight
Place exactly 10 pieces of that bin's cutlery type on the scale. The raw reading ÷ known weight (10 × piece grams) = your calibration factor. Update CALIBRATION[] in firmware.
Step 3 — Verify
Add and remove pieces. Confirm the count matches within ±1 piece. The HX711 at 24-bit resolution is accurate to fractions of a gram — variation comes from pieces not being perfectly identical weight.
Tip: Weigh 20 pieces of each type individually and use the average. For mixed bins (e.g. salad forks and dinner forks together) you'll want separate bins per type — the sensor can't distinguish pieces by type, only by total weight.
