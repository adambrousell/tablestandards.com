01

Toast API pull pull_pmix.py — both paths

This script is identical regardless of which storage backend you choose. It authenticates against Toast, pulls item-selection detail for the last 7 days, and normalizes the data into a clean DataFrame that the next step consumes.

Toast API field names Toast renames fields occasionally between API versions. Run a test pull and print(df.columns.tolist()) to confirm your exact field names before wiring the rename map below.
pull_pmix.py 
import os, requests, pandas as pd
from datetime import datetime, timedelta
from dotenv import load_dotenv

load_dotenv()

TOAST_HOST      = "https://ws-api.toasttab.com"
CLIENT_ID       = os.environ["TOAST_CLIENT_ID"]
CLIENT_SECRET   = os.environ["TOAST_CLIENT_SECRET"]
RESTAURANT_GUID = os.environ["TOAST_RESTAURANT_GUID"]


def get_token() -> str:
    r = requests.post(
        f"{TOAST_HOST}/authentication/v1/authentication/login",
        json={
            "clientId":      CLIENT_ID,
            "clientSecret":  CLIENT_SECRET,
            "userAccessType":"TOAST_MACHINE_CLIENT",
        },
    )
    r.raise_for_status()
    return r.json()["token"]["accessToken"]


def pull_pmix(start_dt: datetime, end_dt: datetime) -> pd.DataFrame:
    token   = get_token()
    headers = {
        "Authorization":                f"Bearer {token}",
        "Toast-Restaurant-External-ID": RESTAURANT_GUID,
    }
    params = {
        "startDate": start_dt.strftime("%Y%m%d"),
        "endDate":   end_dt.strftime("%Y%m%d"),
    }

    r = requests.get(
        f"{TOAST_HOST}/reporting/v1/reports/itemselectiondetails",
        headers=headers, params=params,
    )
    r.raise_for_status()

    df = pd.DataFrame(r.json())
    df = df.rename(columns={
        "menuItemName":  "item_name",
        "menuGroupName": "category",
        "quantity":      "qty_sold",
        "grossSales":    "gross_sales",
        "netSales":      "net_sales",
        "orderDate":     "order_date",
    })

    df["order_date"] = pd.to_datetime(df["order_date"])
    df["week_start"] = (
        df["order_date"] - pd.to_timedelta(df["order_date"].dt.dayofweek, unit="d")
    )
    df["gross_sales"] = df["gross_sales"] / 100  # cents → dollars
    df["net_sales"]   = df["net_sales"]   / 100

    return df[[
        "item_name", "category", "qty_sold",
        "gross_sales", "net_sales", "order_date", "week_start",
    ]]


if __name__ == "__main__":
    end   = datetime.now()
    start = end - timedelta(days=7)
    df    = pull_pmix(start, end)
    print(df.head(10))
    print(f"\n{len(df)} rows pulled")
    df.to_csv("pmix_latest.csv", index=False)  # always write CSV fallback

SQLite load load_pmix.py

Idempotent upsert — deletes the loaded date range then re-inserts, so re-running never duplicates rows. Zero infrastructure: a single pmix.db file on disk. Skip this section entirely if you're using the Airtable path.

load_pmix.py 
import sqlite3, pandas as pd
from pathlib import Path

DB_PATH = Path("pmix.db")

CREATE_SQL = """
CREATE TABLE IF NOT EXISTS pmix (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    item_name   TEXT    NOT NULL,
    category    TEXT,
    qty_sold    REAL    DEFAULT 0,
    gross_sales REAL    DEFAULT 0,
    net_sales   REAL    DEFAULT 0,
    order_date  TEXT    NOT NULL,
    week_start  TEXT    NOT NULL
)
"""


def upsert_pmix(df: pd.DataFrame) -> int:
    con = sqlite3.connect(DB_PATH)
    con.execute(CREATE_SQL)
    con.execute("CREATE INDEX IF NOT EXISTS idx_dates ON pmix(week_start, order_date)")

    min_date = df["order_date"].min()
    max_date = df["order_date"].max()
    if isinstance(min_date, pd.Timestamp):
        min_date = min_date.date().isoformat()
        max_date = max_date.date().isoformat()

    con.execute("DELETE FROM pmix WHERE order_date BETWEEN ? AND ?", (min_date, max_date))

    df = df.copy()
    df["order_date"] = df["order_date"].dt.date.astype(str)
    df["week_start"] = df["week_start"].dt.date.astype(str)

    df.to_sql("pmix", con, if_exists="append", index=False)
    con.commit(); con.close()
    return len(df)


if __name__ == "__main__":
    from pull_pmix import pull_pmix
    from datetime import datetime, timedelta
    end = datetime.now(); start = end - timedelta(days=7)
    n = upsert_pmix(pull_pmix(start, end))
    print(f"✓ {n} rows loaded to {DB_PATH}")

Airtable schema setup airtable path

Create a new table called PMix in your existing Airtable base (you can use the same base as the MASI inspector portal or create a dedicated River Pointe base). Set up these fields manually before running the push script.

Row limits Airtable free tier caps at 1,000 rows per base. A year of daily PMix data for a busy restaurant can hit 50k+ rows — you'll want a Team plan (50,000 rows/base) or a dedicated PMix base with periodic archiving of old weeks to keep row count down.

2B

Push to Airtable airtable path

Replaces load_pmix.py in the SQLite path. Airtable's API accepts records in batches of up to 10. The script deletes existing records for the date window first (idempotent), then inserts the fresh batch. A CSV is always written alongside as a fallback for Power BI.

Rate limit Airtable enforces 5 requests per second per base. The script includes a short time.sleep(0.25) between batches to stay safely under the limit. For a week of data this adds at most a few seconds.
load_airtable.py 
import os, time, requests, pandas as pd
from dotenv import load_dotenv

load_dotenv()

AIRTABLE_TOKEN   = os.environ["AIRTABLE_TOKEN"]
AIRTABLE_BASE_ID = os.environ["AIRTABLE_BASE_ID"]
AIRTABLE_TABLE   = os.environ.get("AIRTABLE_PMIX_TABLE", "PMix")

AT_BASE = (
    f"https://api.airtable.com/v0/{AIRTABLE_BASE_ID}"
    f"/{AIRTABLE_TABLE}"
)
HEADERS = {
    "Authorization": f"Bearer {AIRTABLE_TOKEN}",
    "Content-Type":  "application/json",
}


def delete_week(week_start: str) -> int:
    """Delete all existing records for a given week_start date string."""
    deleted = 0
    offset  = None

    while True:
        params = {
            "filterByFormula": f"{{Week Start}}='{week_start}'",
            "fields[]": "Week Start",
        }
        if offset:
            params["offset"] = offset

        r = requests.get(AT_BASE, headers=HEADERS, params=params)
        r.raise_for_status()
        data    = r.json()
        records = data.get("records", [])

        if not records:
            break

        # Delete up to 10 at a time
        ids = [rec["id"] for rec in records]
        for i in range(0, len(ids), 10):
            params_del = {"records[]": ids[i:i+10]}
            rd = requests.delete(AT_BASE, headers=HEADERS, params=params_del)
            rd.raise_for_status()
            deleted += len(ids[i:i+10])
            time.sleep(0.25)

        offset = data.get("offset")
        if not offset:
            break

    return deleted


def push_to_airtable(df: pd.DataFrame) -> int:
    """Idempotent push: delete existing week, then insert fresh records."""
    # Get unique week_starts to clean before inserting
    week_starts = df["week_start"].dt.date.astype(str).unique()
    for ws in week_starts:
        n_del = delete_week(ws)
        if n_del:
            print(f"  Deleted {n_del} existing records for week {ws}")

    # Build record list
    records = []
    for _, row in df.iterrows():
        records.append({
            "fields": {
                "Item Name":   row["item_name"],
                "Category":    row["category"],
                "Qty Sold":    float(row["qty_sold"]),
                "Gross Sales": float(row["gross_sales"]),
                "Net Sales":   float(row["net_sales"]),
                "Order Date":  row["order_date"].date().isoformat(),
                "Week Start":  row["week_start"].date().isoformat(),
            }
        })

    # Batch insert (max 10 per request)
    created = 0
    for i in range(0, len(records), 10):
        batch = records[i:i+10]
        r = requests.post(AT_BASE, headers=HEADERS, json={"records": batch})
        r.raise_for_status()
        created += len(r.json().get("records", []))
        time.sleep(0.25)  # stay under 5 req/sec rate limit

    return created


def export_csv_from_airtable(output_path: str = "pmix_latest.csv") -> None:
    """Pull all records from Airtable and write a CSV for Power BI."""
    all_records = []
    offset = None

    while True:
        params = {"view": "Grid view"}
        if offset:
            params["offset"] = offset
        r = requests.get(AT_BASE, headers=HEADERS, params=params)
        r.raise_for_status()
        data = r.json()
        all_records.extend(data.get("records", []))
        offset = data.get("offset")
        if not offset:
            break
        time.sleep(0.25)

    rows = [rec["fields"] for rec in all_records]
    df   = pd.DataFrame(rows)
    df.to_csv(output_path, index=False)
    print(f"  ✓ Exported {len(df)} rows → {output_path}")


if __name__ == "__main__":
    from pull_pmix import pull_pmix
    from datetime import datetime, timedelta
    end   = datetime.now()
    start = end - timedelta(days=7)
    df    = pull_pmix(start, end)
    n     = push_to_airtable(df)
    print(f"✓ {n} records in Airtable")
    export_csv_from_airtable()  # refresh CSV for Power BI

Power BI + Airtable airtable path

Airtable has no native Power BI connector. The recommended approach is to have the pipeline write a fresh pmix_latest.csv (the export_csv_from_airtable() call at the end of the script), then point Power BI at that CSV via the on-prem gateway. This also means Airtable serves as the human-readable live view and CSV is the Power BI data source — you get both.

Option A — CSV bridge (recommended)
  • Pipeline writes CSV after each Airtable push
  • Power BI reads CSV via on-prem gateway
  • Zero extra dependencies
  • Staff can view live data in Airtable
  • Power BI gets a clean weekly snapshot
Option B — Third-party connector
  • Coupler.io or Supermetrics sync Airtable → PBI
  • No CSV file needed
  • Paid dependency ($20–50/mo)
  • Scheduled sync — not real-time
  • Adds a third service to maintain

DAX measures add in Power BI Desktop

Create these as explicit measures (Home → New Measure). They power all four dashboard pages regardless of storage backend.

Qty This Week total qty in current week filter
Qty This Week =
CALCULATE(
    SUM(pmix[qty_sold]),
    DATESINPERIOD(pmix[week_start], LASTDATE(pmix[week_start]), -1, WEEK)
)
Category Mix % item share of total qty
Category Mix % =
DIVIDE(
    SUM(pmix[qty_sold]),
    CALCULATE(SUM(pmix[qty_sold]), ALL(pmix[category]))
)
Revenue per Item average selling price
Revenue per Item =
DIVIDE(SUM(pmix[gross_sales]), SUM(pmix[qty_sold]))
QoW Change % week-over-week qty movement
QoW Change % =
VAR thisWeek = SUM(pmix[qty_sold])
VAR lastWeek = CALCULATE(
    SUM(pmix[qty_sold]),
    DATEADD(pmix[week_start], -1, WEEK)
)
RETURN DIVIDE(thisWeek - lastWeek, lastWeek)
Gross Sales This Week dollar total, current week
Gross Sales This Week =
CALCULATE(
    SUM(pmix[gross_sales]),
    DATESINPERIOD(pmix[week_start], LASTDATE(pmix[week_start]), -1, WEEK)
)
Top N Filter dynamic top-N via What-If parameter
-- 1. Add What-If parameter: TopN, 1-50, default 20
-- 2. Use this measure as a visual-level filter > 0

Top N Filter =
VAR rank = RANKX(
    ALL(pmix[item_name]),
    CALCULATE(SUM(pmix[qty_sold])),
    , DESC, Dense
)
RETURN IF(rank <= [TopN Value], 1, 0)

Export & email send_report.py

Authenticates against the Power BI REST API via Azure AD service principal, kicks off a PDF export job, polls until complete, then emails the PDF via SMTP. Works with any SMTP provider — Gmail app passwords are the fastest path.

Azure App Registration Create an Azure AD app registration with the Power BI API permission Report.ReadAll. Grant admin consent. Takes ~10 minutes in the Azure portal. Not needed if you export PDF manually from Power BI Desktop.
send_report.py 
import os, smtplib, time, requests
from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email.mime.text import MIMEText
from email import encoders
from datetime import datetime
from dotenv import load_dotenv

load_dotenv()

TENANT_ID     = os.environ["AZURE_TENANT_ID"]
CLIENT_ID     = os.environ["AZURE_CLIENT_ID"]
CLIENT_SECRET = os.environ["AZURE_CLIENT_SECRET"]
WORKSPACE_ID  = os.environ["PBI_WORKSPACE_ID"]
REPORT_ID     = os.environ["PBI_REPORT_ID"]
SMTP_HOST     = os.environ["SMTP_HOST"]
SMTP_PORT     = int(os.environ.get("SMTP_PORT", 587))
SMTP_USER     = os.environ["SMTP_USER"]
SMTP_PASS     = os.environ["SMTP_PASS"]

RECIPIENTS = [
    "gm@yourrestaurant.com",
    "chef@yourrestaurant.com",
    "ownership@yourgroup.com",
]

PBI_BASE = (
    f"https://api.powerbi.com/v1.0/myorg/groups/{WORKSPACE_ID}"
    f"/reports/{REPORT_ID}"
)


def get_pbi_token() -> str:
    r = requests.post(
        f"https://login.microsoftonline.com/{TENANT_ID}/oauth2/v2.0/token",
        data={
            "grant_type":    "client_credentials",
            "client_id":     CLIENT_ID,
            "client_secret": CLIENT_SECRET,
            "scope": "https://analysis.windows.net/powerbi/api/.default",
        },
    )
    r.raise_for_status()
    return r.json()["access_token"]


def export_report_pdf(token: str) -> bytes:
    headers = {"Authorization": f"Bearer {token}"}
    r = requests.post(f"{PBI_BASE}/ExportTo", headers=headers, json={"format": "PDF"})
    r.raise_for_status()
    export_id = r.json()["id"]
    print(f"  Export id: {export_id}")

    for attempt in range(30):
        time.sleep(5)
        sr    = requests.get(f"{PBI_BASE}/exports/{export_id}", headers=headers)
        st    = sr.json().get("status")
        pct   = sr.json().get("percentComplete", 0)
        print(f"  [{attempt+1}/30] {st} ({pct}%)")
        if st == "Succeeded":
            fr = requests.get(f"{PBI_BASE}/exports/{export_id}/file", headers=headers)
            return fr.content
        if st == "Failed":
            raise RuntimeError(f"Export failed: {sr.json()}")

    raise TimeoutError("Power BI export timed out")


def send_email(pdf_bytes: bytes, week_label: str) -> None:
    msg = MIMEMultipart()
    msg["From"]    = SMTP_USER
    msg["To"]      = ", ".join(RECIPIENTS)
    msg["Subject"] = f"Weekly PMix Report — {week_label}"
    msg.attach(MIMEText(f"""Team,

Attached is the product mix report for the week of {week_label}.

  • Top 20 items by qty sold (page 1)
  • Category revenue mix treemap (page 2)
  • Week-over-week movement (page 4)

Generated automatically every Monday from live Toast POS data.
"""))
    att = MIMEBase("application", "octet-stream")
    att.set_payload(pdf_bytes)
    encoders.encode_base64(att)
    att.add_header("Content-Disposition", f'attachment; filename="PMix_{week_label}.pdf"')
    msg.attach(att)

    with smtplib.SMTP(SMTP_HOST, SMTP_PORT) as server:
        server.starttls()
        server.login(SMTP_USER, SMTP_PASS)
        server.sendmail(SMTP_USER, RECIPIENTS, msg.as_string())
    print(f"  ✓ Emailed to {len(RECIPIENTS)} recipients")


if __name__ == "__main__":
    week = datetime.now().strftime("%Y-%m-%d")
    token = get_pbi_token()
    pdf   = export_report_pdf(token)
    send_email(pdf, week)
    print("Done.")
